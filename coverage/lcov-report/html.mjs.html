/**

 * @param {Node} node
 * @param {(node: Element) => boolean} predicate
 * @returns {Element[]}
 */
export function findAll(node, predicate) {
  const found = [];
  const visit = (current) => {
    if (current.type !== "element") return;
    if (predicate(current)) found.push(current);
    for (const child of current.children) visit(child);
  };
  visit(node);
  return found;
}
 
/**
